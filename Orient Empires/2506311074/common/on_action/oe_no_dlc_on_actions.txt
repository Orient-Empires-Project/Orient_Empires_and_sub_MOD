on_game_start = {
	on_actions = {
		celes_no_dlc_on_action
		ulus_no_dlc_on_action
	}
}

celes_no_dlc_on_action = {
    effect = {
		if = {
			limit = {
				NOT = { has_dlc_feature = all_under_heaven }
			}
			every_ruler = {
				celestial_change_to_imperial_with_laws_effect = yes
			}
		}
        if = {
            limit = {
                has_dlc_feature = roads_to_power
            }
			title:e_byzantium.holder = {
                if = {
                    limit = {
                        NOT = { government_has_flag = government_is_administrative }
                    }
                    change_government = administrative_government
                    add_realm_law_skip_effects = single_heir_succession_law
                }
            }
        }
		else = {
			title:e_byzantium.holder = {
				change_government = feudal_government
				add_realm_law_skip_effects = single_heir_succession_law
			}
		}
	}
}

ulus_no_dlc_on_action = {
	effect = {
		if = {
			limit = {
				has_dlc_feature = all_under_heaven
			}
			every_ruler = {
				limit = { government_has_flag = government_is_ulus }
				if = {
					limit = {
						capital_province ?= {
							OR = {
								has_holding_type = nomad_holding
								has_holding_type = tribal_holding
							}
						}
					}
					capital_province = {
						set_holding_type = castle_holding
					}
					change_government = steppe_admin_government
				}
			}
		}
		else_if = {
			limit = {
				has_dlc_feature = khans_of_the_steppe
			}
			every_ruler = {
				limit = {
					OR = {
						government_has_flag = government_is_ulus
						government_has_flag = government_is_steppe_admin
					}
				}
				if = {
					limit = {
						capital_province ?= {
							OR = {
								has_holding_type = nomad_holding
								has_holding_type = tribal_holding
							}
						}
					}
					capital_province = {
						set_holding_type = nomad_holding
					}
					change_government = nomad_government
					if = {
						limit = { top_liege != this }
						vassal_contract_set_obligation_level = { type = liege_war_participation_obligation level = 1 }
					}

					every_vassal_or_below = {
						limit = {
							is_landed = yes
							capital_province ?= {
								has_holding_type = castle_holding
							}
							OR = {
								has_government = celestial_government
								has_government = feudal_government
								has_government = ulus_government
							}
						}
						change_government = stratocratic_government
					}

					trigger_event = {
						on_action = ulus_change_nomad
						days = 2
					}
				}
				else = {
					change_government = feudal_government
				}
			}
		}

		#oe_setup_tributaries_effect = yes

		if = {
            limit = {
                NOT = { has_dlc_feature = khans_of_the_steppe }
            }
			every_ruler = {
				limit = {
					government_has_flag = government_is_steppe_admin
				}
				change_government = ulus_government
			}
			every_county_province = {
				limit = {
					OR = {
						has_holding_type = nomad_holding
						has_holding_type = herder_holding
					}
				}
				set_holding_type = tribal_holding
				county.holder = {
					change_government = tribal_government
					if = {
						limit = {
							highest_held_title_tier >= tier_empire
						}
						add_realm_law_skip_effects = tribal_authority_3
					}
					else_if = {
						limit = {
							highest_held_title_tier = tier_kingdom
						}
						add_realm_law_skip_effects = tribal_authority_2
					}
					else = {
						add_realm_law_skip_effects = tribal_authority_1
					}
				}
			}
        }
    }
}

ulus_change_nomad = {
	effect = {
		ulus_change_nomad_effect = yes
	}
}
