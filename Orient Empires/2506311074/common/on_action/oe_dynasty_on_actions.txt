oe_merit_on_action = {
	effect = {
		every_ruler = {
			if = {
				limit = {
					OR = {
						title:e_zhongyuan.holder ?= top_liege
						title:e_liangyi.holder ?= top_liege
						title:e_jingyang.holder ?= top_liege
						title:e_yongliang.holder ?= top_liege
						title:e_lingnan.holder ?= top_liege
						title:e_andong.holder ?= top_liege

						title:h_china.holder ?= top_liege
					}
					this != top_liege
				}
				### TGP MERIT AND TREASURY HANDOUT ON GAME START ###
				if = {
					limit = {
						culture = {
							has_cultural_pillar = heritage_chinese
						}
						NOR = {
							AND = {
								has_council_position = councillor_court_chaplain
								NOT = { has_title = title:e_minister_of_rites }
							}
							has_trait = devoted
						}
						government_has_flag = government_has_merit
						is_faith_dominant_gender = yes
					}
					save_temporary_scope_as = exam_contender

					switch = {
						trigger = highest_held_title_tier
						tier_empire = {
							change_merit = {
								value = empire_starting_merit_value
							}
						}
						tier_kingdom = {
							change_merit = {
								value = kingdom_starting_merit_value
							}
						}
						tier_duchy = {
							change_merit = {
								value = duchy_starting_merit_value
							}
						}
						tier_county = {
							change_merit = {
								value = county_starting_merit_value
							}
						}
					}
					if = {
						limit = {
							has_trait = zhuangyuan
						}
						change_merit = {
							value = kingdom_starting_merit_value
						}
					}
					else_if = {
						limit = {
							has_trait = jinshi
						}
						change_merit = {
							value = duchy_starting_merit_value
						}
					}
					if = {
						limit = {
							is_house_head = yes
							dynasty = {
								dynasty_prestige_level > 0
							}
						}
						change_merit = {
							value = 1000
							multiply = dynasty.dynasty_prestige_level
						}
					}

					hidden_effect = {
						assign_completed_exams_based_on_merit_effect = yes
					}
					if = {
						limit = {
							any_parent = {
								OR = {
									has_trait = ambitious
									has_trait = scholar
									has_trait = diligent
									has_trait = confucian_education
								}
							}
							NOR = {
								has_trait = lazy
								has_trait = dull
								has_trait = intellect_bad
								location.county ?= {
									development_level <= bad_development_level
								}
							}
						}
						add_character_flag = passed_child_exam
					}
					#if = { # We adjust Merit depending on the dynastic cycle phase
						#limit = {
							#situation:dynastic_cycle ?= {
								#situation_current_phase = situation_dynastic_cycle_phase_stability_advancement
							#}
						#}
						#change_merit = {
							#value = merit
							#multiply = 0.1
						#}
					#}
					
					if = {
						limit = {
							is_ai = no
							is_governor_or_admin_count = yes
						}
						spawn_celestial_governance_contract_effect = yes
					}
				}
				### Examination pass status and initial wave of governor candidates done after game start, in order to include generated families ###
				every_courtier = {
					limit = {
						is_lowborn = no
						culture = { has_cultural_pillar = heritage_chinese }
						NOR = {
							has_council_position = councillor_court_chaplain
							has_trait = devoted
						}
						OR = {
							government_has_flag = government_is_celestial # courtiers have governments, too!
							government_has_flag = government_is_stratocratic
						}
						
						is_faith_dominant_gender = yes
						age >= 30
						static_group_filter = {
							group = ai_filter
							match = 0.5
						}
					}
					hidden_effect = {
						add_character_flag = passed_provincial_exam
						add_character_modifier = {
							modifier = tgp_passed_provincial_exam_modifier
						}
						add_character_flag = passed_metropolitan_exam
						add_character_modifier = {
							modifier = tgp_passed_metropolitan_exam_modifier
						}
						change_merit = {
							value = county_starting_merit_value
							add = age
							add = learning
						}
					}
				}
				save_scope_as = governor
			}
		}
		
	}
}

#世族
oe_noble_family_on_action = {
	effect = {
		title:h_china.holder ?= {
			add_to_list = list_chinese_noble_family_head
			every_vassal_or_below = {
				limit = {
					is_house_head = yes
				}
				add_to_list = list_chinese_noble_family_head
			}
			every_courtier = {
				limit = {
					is_house_head = yes
				}
				add_to_list = list_chinese_noble_family_head
			}
		}
		every_independent_ruler = {
			limit = {
				government_use_bureaucracy_trigger = yes
				highest_held_title_tier >= tier_duchy
			}
			add_to_list = list_chinese_noble_family_head
			every_vassal_or_below = {
				limit = {
					is_house_head = yes
				}
				add_to_list = list_chinese_noble_family_head
			}
			every_courtier = {
				limit = {
					is_house_head = yes
				}
				add_to_list = list_chinese_noble_family_head
			}
		}
		every_in_list = {
			list = list_chinese_noble_family_head
			
			give_new_noble_family_title_effect = yes
			dynasty = {
				add_dynasty_prestige_level = 2
			}
		}
	}
}

great_dynasty_marriage = {
	effect = {
		root = {
			if = { # Non-lowborn Barons should marry and produce a family
				limit = {
					exists = dynasty
					culture = { has_cultural_pillar = heritage_chinese }
					#location.county.holder.top_liege = {
						#government_allows = administrative
					#}
					dynasty = { dynasty_prestige_level >= 3 }	
					is_adult = yes
					is_married = no
					can_marry_trigger = yes
					patrilinear_betrothal = no
					matrilinear_betrothal = no
					is_concubine = no
					is_ai = yes
					is_ruler = no
					NAND = {
						exists = matchmaker
						matchmaker = {
							OR = {
								is_ai = no
								this = liege
								highest_held_title_tier >= tier_county
							}
						}
					}
				}
				
				if = {
					limit = {
						exists = root.capital_province
						any_pool_character = {
							province = root.capital_province
							can_have_children_with = { CHARACTER = root }
							is_physically_able_adult = yes
							age <= 40
							is_married = no
							can_marry_trigger = yes
							is_concubine = no
							culture = root.culture
						}
					}
					random_pool_character = {
						province = root.capital_province
						limit = {							
							can_have_children_with = { CHARACTER = root }
							is_physically_able_adult = yes
							age <= 40
							is_married = no
							can_marry_trigger = yes
							is_concubine = no
							culture = root.culture
						}
						save_temporary_scope_as = spouse
					}
				}
				else = {
					create_character = {
						location = root.location
						culture = root.culture
						template = pool_repopulate_spouse
						save_temporary_scope_as = spouse
					}
				}

				if = {
					limit = {
						is_female = yes
						scope:spouse = {
							is_lowborn = yes
						}
					}
					marry_matrilineal = scope:spouse
				}
				else = {
					marry = scope:spouse
				}
			}
		}
	}
}
