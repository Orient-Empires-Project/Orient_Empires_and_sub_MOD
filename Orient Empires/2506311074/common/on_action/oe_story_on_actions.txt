	}

oe_story_on_death = {
	effect = {
		if = {
			limit = {
				game_start_date = 925.11.14 #如梦令
				liege = primary_title.holder
				exists = scope:killer
			}
			primary_title.holder = {
				trigger_event = {
					on_action = chinese_revolt_events_on_action
				}
				every_powerful_vassal = {
					add_opinion = {
						target = prev
						modifier = executed_close_relation_opinion
					}
					trigger_event = {
						on_action = chinese_revolt_events_on_action
					}
				}
			}
		}
		#if = {
			#limit = { has_title = title:d_incho } #销毁院厅
			#destroy_title = title:d_incho
		#}
		if = {
			limit = {
				game_start_date = 617.8.1 #长歌行
				#primary_title.holder ?= character:247726
				root = character:247726
			}
			if = {
				limit = {
					exists = scope:killer
					scope:killer = title:e_minister_grand_marshal.holder
				}
				if = {
					limit = {
						character:247755 = { is_alive = yes }
					}
					create_title_and_vassal_change = {
						type = usurped
						save_scope_as = change
						add_claim_on_loss = no
					}
					primary_title = {
						change_title_holder = {
							holder = character:247755
							change = scope:change
						}
					}
					resolve_title_and_vassal_change = scope:change
				}
				else_if = {
					limit = {
						character:247761 = { is_alive = yes }
					}
					create_title_and_vassal_change = {
						type = usurped
						save_scope_as = change
						add_claim_on_loss = no
					}
					primary_title = {
						change_title_holder = {
							holder = character:247761
							change = scope:change
						}
					}
					resolve_title_and_vassal_change = scope:change
				}
				else = {
					title:e_minister_grand_marshal.holder = {
						claim_fake_chinese_emperor_effect = yes
						primary_title = {
							set_title_name = root.capital_province.kingdom
						}
					}
				}
			}
			every_house_member = {
				limit = {
					NOT = { location = root.location }
					highest_held_title_tier >= tier_kingdom
				}
				claim_fake_chinese_emperor_effect = yes
				primary_title = {
					set_title_name = e_sui
				}
			}
			primary_title.holder = {
				trigger_event = {
					on_action = chinese_revolt_events_on_action
				}
				every_powerful_vassal = {
					add_opinion = {
						target = prev
						modifier = executed_close_relation_opinion
					}
					trigger_event = {
						on_action = chinese_revolt_events_on_action
					}
				}
			}
		}
	}
}

on_title_gain = {
	on_actions = {
		oe_story_on_title_gain
	}
}

oe_story_on_title_gain = {
	effect = {
		#长歌行
		if = {
			limit = {
				exists = global_var:sui_era
				scope:title = primary_title
				NOT = {
					dynasty ?= dynasty:hongnong_yang_dyn
				}
			}
			remove_global_variable = sui_era
		}
		#关山月
		if = {
			limit = {
				exists = global_var:tianbao_era
				scope:title = primary_title
				NAND = {
					dynasty ?= dynasty:longxi_li_dyn
					this = character:longxi_li_0952
				}
			}
			remove_global_variable = tianbao_era
		}
		#如梦令
		if = {
			limit = {
				game_start_date >= 925.1.1
				game_start_date < 947.12.12
			}
			if = {
				limit = {
					scope:transfer_type = flag:conquest
					scope:title = { tier >= tier_empire }
					NOT = { scope:title = primary_title }
					has_title = title:e_minister_grand_marshal
				}
				destroy_title = scope:title
			}
			#枢密院入川
			if = {
				limit = {
					scope:transfer_type = flag:conquest
					scope:title = title:k_shu
					root = character:1401816
					title:e_minister_grand_marshal.holder ?= character:1401816
					liege = primary_title.holder
				}
				#唐皇子谋杀西蜀军头
				primary_title.holder.primary_heir = {
					start_scheme = {
						type = murder
						target = title:e_minister_grand_marshal.holder
					}
				}
				primary_title.holder.primary_spouse = {
					start_scheme = {
						type = murder
						target = title:e_minister_grand_marshal.holder
					}
				}
				primary_title.holder = {
					random_close_family_member = {
						start_scheme = {
							type = murder
							target = title:e_minister_grand_marshal.holder
						}
					}
				}
			}
			##移交河东军
			if = {
				limit = {
					scope:transfer_type = flag:inheritance
					OR = {
						scope:title = title:e_minister_grand_marshal
						scope:title = title:d_xingfeng
						scope:title = title:k_shu
					}
					scope:previous_holder = character:1401816
					root = character:longgang_meng_06
					top_liege != this
				}
				character:longgang_meng_06 = {
					liege = {
						save_temporary_scope_as = liege
					}
				}
				every_held_title = {
					limit = {
						NOT = { title_capital_county.title_province.kingdom = title:k_shu }
					}
					add_to_list = titles_to_take
				}
				every_vassal = {
					limit = {
						NOT = { capital_province.kingdom = title:k_shu }
					}
					add_to_list = vassal_to_take
				}
				create_title_and_vassal_change = {
					type = usurped
					save_scope_as = change
					add_claim_on_loss = no
				}
				every_in_list = {
					list = titles_to_take
					change_title_holder_include_vassals = {
						holder = scope:liege
						change = scope:change
						take_baronies = no
					}
				}
				every_in_list = {
					list = vassal_to_take
					change_liege = {
						liege = scope:liege
						change = scope:change
					}
				}
				resolve_title_and_vassal_change = scope:change
			}
		}
	}
