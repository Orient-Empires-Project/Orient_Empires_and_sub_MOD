on_title_gain = {
	on_actions = {
		miaohao_on_title_gain
	}
}

miaohao_on_title_gain = {
	trigger = {
		scope:title.tier >= tier_empire
		NOT = { house = scope:title.previous_holder.house }
		OR = {
			scope:transfer_type = flag:created
			scope:transfer_type = flag:usurped
			scope:transfer_type = flag:conquest_claim
		}
		#帝国升天朝不重复获得
		NAND = {
			scope:title.tier > tier_empire 
			any_held_title = {
				tier = tier_empire
			}
			capital_province ?= { geographical_region = world_asia_china }
		}
	}
	effect = {
		if = {
			limit = {
				highest_held_title_tier >= tier_empire
				royal_members_trigger = yes ##前皇族
				NOR = {
					has_trait = founding_monarch
					has_trait = restoration_monarch
				}
			}
			add_trait = restoration_monarch
		}
		else_if = {
			limit = {
				highest_held_title_tier >= tier_empire
				royal_members_trigger = no ##非皇族
				NOT = { has_trait = founding_monarch }
			}
			father = {
				if = {
					limit = {
						is_alive = no
					}
					add_trait = miaohao_gaozu
				}
			}
			add_trait = founding_monarch
		}
	}
}
