#Triggers to control Coat of Arms weighting

coa_chinese_trigger = {
	scope:culture = { has_coa_gfx = chinese_group_coa_gfx }
}

coa_chinese_religion_trigger = {
	OR = {
		has_religion = religion:taoism_religion
		has_religion = religion:confucianism_religion
	}
}

coa_chinese_region_trigger = {
	capital_province ?= { geographical_region = world_asia_china }
}


coa_yamato_trigger = {
	scope:culture = {
		OR = {
			has_coa_gfx = japanese_coa_gfx
			has_coa_gfx = ryukyu_group_coa_gfx
		}
	}
}

coa_yamato_faith_trigger = {
	OR = {
		scope:faith.religion = religion:shintoism_religion
		scope:faith.religion = religion:taoism_religion
		scope:faith.religion = religion:confucianism_religion
	}
}

coa_ryukyu_trigger = {
	scope:culture = { has_coa_gfx = ryukyu_group_coa_gfx }
}

coa_ainu_trigger = {
	scope:culture = { has_coa_gfx = ainu_group_coa_gfx }
}


coa_eastern_colors_trigger = {
	OR = {
		coa_chinese_trigger = yes
		coa_yamato_trigger = yes
		coa_yamato_faith_trigger = yes
		coa_ainu_trigger = yes		
		coa_austronesian_trigger = yes
		#coa_ifugao_trigger = yes
		coa_nuosu_bai_trigger = yes
		coa_tai_trigger = yes
		coa_zhuang_trigger = yes
		#coa_java_trigger = yes
		#coa_sulawesi_trigger = yes
		#coa_malay_trigger = yes
		#coa_filipino_trigger = yes
		#coa_mon_trigger = yes
		#coa_dayak_trigger = yes
		#coa_tibetan_trigger = yes
	}
}

coa_austronesian_trigger = {
	scope:culture = { has_coa_gfx = austronesian_group_coa_gfx }
}

coa_ifugao_trigger = {
	scope:culture = { has_coa_gfx = ifugao_coa_gfx }
}

coa_nuosu_bai_trigger = {
	scope:culture = { has_coa_gfx = lolo_group_coa_gfx }
}


coa_tai_trigger = {
	scope:culture = { has_coa_gfx = tai_group_coa_gfx }
}

coa_zhuang_trigger = {
	scope:culture = { has_coa_gfx = zhuang_group_coa_gfx }
}
