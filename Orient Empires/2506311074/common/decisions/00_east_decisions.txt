found_new_chinese_empire_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_activity.dds"
	}
  	decision_group_type = imperial
	ai_goal = yes

	is_shown = {
		is_ai = no
	 	is_landed = yes
		is_independent_ruler = yes
		highest_held_title_tier = tier_kingdom
		government_use_bureaucracy_trigger = yes
		capital_province ?= {
			geographical_region = world_asia_china
		}
		NOT = { has_title = title:h_china }		
	}

	is_valid = {
		prestige_level >= 2
		trigger_if = {
			limit = {
				NOT = { exists = title:h_china.holder }
			}
			OR = {
				capital_province.empire = { holder = root }
				capital_province.empire = {
					any_in_de_jure_hierarchy = {
						tier = tier_kingdom
						holder = root
					}
				}
				AND = {
					capital_province.empire.title_capital_county = { holder = root }
					capital_province.empire.title_capital_county.kingdom = { holder = root }
				}

				calc_true_if = {
					amount >= 5
					completely_controls = title:c_jingzhao
					completely_controls = title:c_henan
					completely_controls = title:c_bianzhou
					completely_controls = title:c_taiyuan
					completely_controls = title:c_weizhou
					completely_controls = title:k_wu
					completely_controls = title:k_yue
					has_title = title:k_chu
					has_title = title:k_shu
					has_title = title:k_lingnan
				}
			}
		}
		trigger_if = {
			limit = {
				exists = title:h_china.holder
			}
			OR = {
				capital_province.empire = { holder = root }
				capital_province.empire = {
					any_in_de_jure_hierarchy = {
						tier = tier_kingdom
						holder = root
					}
				}
				AND = {
					capital_province.empire.title_capital_county = { holder = root }
					capital_province.empire.title_capital_county.kingdom = { holder = root }
				}
				calc_true_if = {
					amount >= 8
					completely_controls = title:c_jingzhao #雍
					completely_controls = title:c_henan #豫
					completely_controls = title:c_bianzhou #兖
					completely_controls = title:c_taiyuan #并
					completely_controls = title:c_weizhou #冀
					completely_controls = title:c_chengdu #益
					completely_controls = title:c_jiangning #徐
					completely_controls = title:c_hangzhou #扬
					completely_controls = title:c_xiangyang #荆
					completely_controls = title:c_guangzhou_1 #交
				}
			}
			
		}
		OR = {
			sub_realm_size >= 60
			AND = {
				custom_description = {
					text = found_empire_decision_three_kingdoms_held
					any_held_title = {
						count >= 3
						tier = tier_kingdom
						target_is_de_jure_liege_or_above = title:h_china
					}
				}
				sub_realm_size >= 40
			}
		}
	}

	is_valid_showing_failures_only = {
		top_liege = this
		is_capable_adult = yes
		is_at_war = no
	}

	effect = {
		found_new_chinese_empire_effect = yes
		#结算国号
	}

	cost = {
		gold = 1500
		prestige = 1500
		piety = 1300
	}
	ai_check_interval = 0

		base = 120
	}
