show_chinese_king_effect = {

		if = {
			limit = {
				root = {
					government_has_flag = government_has_influence
				}				
			}
			root = {
				## Clear variable always before
				clear_variable_list = $KINGLIST$
				every_vassal = {
					limit = {
						has_title = $KINGTITLE$
					}
					root = {
						add_to_variable_list = {
							name = $KINGLIST$
							target = prev
						}
					}					
				}
			}
		}
		if = {
			limit = {
				root = {
					NOT = {
						government_has_flag = government_has_influence
					}	
					any_liege_or_above = {
						government_has_flag = government_has_influence
					}				
				}
			}
			root = { clear_variable_list = $KINGLIST$ }
			root.liege = {
				every_vassal = {
					limit = {
						has_title = $KINGTITLE$
					}
					root = {
						add_to_variable_list = {
							name = $KINGLIST$
							target = prev
						}
					}					
				}
			}
		}	
}

OE_create_kingdom_effect = {
	play_music_cue_once = { TRACK = mx_cue_ck3_vocal_maintheme }
	scope:actor = {
		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}
		$KINGTITLE$ = {
			change_title_holder = {
				holder = scope:secondary_actor
				change = scope:change
			}	
		}
		scope:secondary_actor = {
			change_liege = {
				liege = scope:actor
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
	}
	$KINGTITLE$ = {
		set_variable = { name = titular_title value = yes }
	}
	$KINGTITLE$.holder = {
		change_government = stratocratic_government
		add_opinion_to_actor_receive_kingdom = yes
	}
}

gained_principality_effect = {
	scope:liege = {
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = change
			add_claim_on_loss = no
		}
		title:$TITLE$ = {
			change_title_holder = {
				holder = scope:employee
				change = scope:change
			}
		}
		scope:employee = {
			change_liege = {
				liege = scope:liege
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
	}
	title:$TITLE$ = {
		set_variable = { name = titular_title value = yes }
	}
	title:$TITLE$.holder = {
		convert_celestial_contract_commissioner = yes
		add_realm_law_skip_effects = tradition_succession_law
		change_to_hereditary_effect = yes

		add_opinion_to_actor_receive_kingdom = yes #添加 王 称号
	}
}

revoked_principality_effect = {
	if = {
		limit = {
			scope:employee = {
				target_is_vassal_or_below = scope:liege
			}
		}
		scope:employee = {
			destroy_title = title:$TITLE$
		}
	}
}
