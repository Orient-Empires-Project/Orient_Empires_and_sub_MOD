set_shihao_effect = {
	#if = {
		#limit = { has_any_nickname = yes }
		#remove_nickname = yes
	#}
	give_nickname = $SHIHAO$
	set_dead_character_variable = { name = shihao value = flag:$SHIHAO$ }
	house = {
		add_to_variable_list = {
			name = shihao_list target = flag:$SHIHAO$
		}
	}
}

set_miaohao_effect = {
	add_trait = miaohao_$MIAOHAO$
	set_dead_character_variable = { name = mianhao value = flag:$MIAOHAO$ }
	house = {
		add_to_variable_list = {
			name = miaohao_list target = flag:miaohao_$MIAOHAO$
		}
	}
}

add_mornach_trait = {
	if = {
		limit = {
			highest_held_title_tier >= tier_empire
			royal_members_trigger = yes ##前皇族
			NOR = {
				has_trait = founding_monarch
				has_trait = restoration_monarch
			}
		}
		add_trait = restoration_monarch
	}
	else_if = {
		limit = {
			highest_held_title_tier >= tier_empire
			royal_members_trigger = no ##非皇族
			NOT = { has_trait = founding_monarch }
		}
		father = {
			if = {
				limit = {
					is_alive = no
				}
				add_trait = miaohao_gaozu
			}
		}
		add_trait = founding_monarch
	}
}
