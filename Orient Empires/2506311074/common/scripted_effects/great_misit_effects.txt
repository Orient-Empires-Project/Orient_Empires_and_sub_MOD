become_great_misit_effect = {
	set_relation_great_misit_liege = liege
}

misit_power_value_effect = {
	add_misit_power = { VALUE = misit_power_sub_realm_bonus }
}

misit_power_prestige_level_value_effect = {
	add_misit_power = { VALUE = misit_power_prestige_bonus }
}

add_misit_power = {
	custom_description = {
		text = add_misit_power
		value = $VALUE$
		if = {
			limit = {
				NOT = { exists = var:misit_power }
				$VALUE$ > 0
			}
			set_variable = {
				name = misit_power
				value = 0
			}
		}
		change_variable = {
			name = misit_power
			add = $VALUE$
		}
		if = {
			limit = {
				liege ?= {
					has_diarchy_active_parameter = diarchy_use_misit_power
				}
			}
			liege = {
				save_scope_as = temp_liege
			}
			if = {
				limit = {
					$VALUE$ >= 200
				}
				update_diarchy_swing_with_perspective_effect = {
					SWING = monumental_sop_swing_diarch_gain
					LIEGE = scope:temp_liege
				}
			}
			else_if = {
				limit = {
					$VALUE$ >= 100
				}
				update_diarchy_swing_with_perspective_effect = {
					SWING = massive_sop_swing_diarch_gain
					LIEGE = scope:temp_liege
				}
			}
			else_if = {
				limit = {
					$VALUE$ >= 0
				}
				update_diarchy_swing_with_perspective_effect = {
					SWING = major_sop_swing_diarch_gain
					LIEGE = scope:temp_liege
				}
			}
			else_if = {
				limit = {
					$VALUE$ < -200
				}
				update_diarchy_swing_with_perspective_effect = {
					SWING = monumental_sop_swing_liege_gain
					LIEGE = scope:temp_liege
				}
			}
			else_if = {
				limit = {
					$VALUE$ < -100
				}
				update_diarchy_swing_with_perspective_effect = {
					SWING = massive_sop_swing_liege_gain
					LIEGE = scope:temp_liege
				}
			}
			else = {
				update_diarchy_swing_with_perspective_effect = {
					SWING = major_sop_swing_liege_gain
					LIEGE = scope:temp_liege
				}
			}
		}
	}
}
