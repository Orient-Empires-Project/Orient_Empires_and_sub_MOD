legion_government = {

	}
	royal_court = any

	fallback = 3

	#primary_heritages = { heritage_japonic }	

	primary_holding = castle_holding
	valid_holdings = { castle_holding tribal_holding }
	required_county_holdings = { castle_holding city_holding church_holding }
	court_generate_spouses = yes
	court_generate_commanders = 3

	ai = {
		use_legends = yes
	}

	government_can_raid_rule
	government_use_bureaucracy

	vassal_contract_group = legion_vassal

	character_modifier = {
		knight_limit = 12
		vassal_limit = 3	
		tax_mult = -0.2
		levy_size = 0.2

		ai_war_chance = 2
		ai_war_cooldown = -0.5
		army_pursuit_mult = 0.2
		army_siege_value_mult = 0.2
		supply_limit_mult = 0.2
		opinion_of_liege = -15

	}

	opinion_of_liege = {
		scope:vassal = {
			if = {
				limit = {
					NOT = {
						is_allied_to = scope:liege
					}
				}
				if = {
					limit = {
						is_powerful_vassal = yes
					}
					value = clan_powerful_vassal_no_alliance_opinion_penalty_value
				}
				else = {
					value = clan_vassal_no_alliance_opinion_penalty_value
				}
			}
		}
	}
	opinion_of_liege_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:vassal = {
						NOT = {
							is_allied_to = scope:liege
						}
						is_powerful_vassal = yes
					}
				}
				desc = "GOVERNMENT_CLAN_NOT_ALLIED_POWERFUL"
			}
			triggered_desc = {
				trigger = {
					scope:vassal = {
						NOT = {
							is_allied_to = scope:liege
						}
						is_powerful_vassal = no
					}
				}
				desc = "GOVERNMENT_CLAN_NOT_ALLIED"
			}
		}
	}

	flags = {
		may_elevate_co_monarch
		
		government_is_legion
		government_is_settled
		government_uses_domicile_but_not_adventurer
		government_uses_domain_limit
	}
	color = { 137 75 19 }
}
