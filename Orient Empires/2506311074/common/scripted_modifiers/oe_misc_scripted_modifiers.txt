oe_courtier_management_leave_chance_modifier  = {
	#高阶流官备选人离开领主宫廷
	modifier = {
		is_courtier_of = scope:liege
		has_trait = lifestyle_bureaucrat
		scope:liege = { has_trait = lifestyle_bureaucrat }
		bureaucrat_level_value > scope:liege.bureaucrat_level_value
		add = 50
	}
	modifier = {
		is_courtier_of = scope:liege
		has_trait = lifestyle_bureaucrat
		scope:liege = { has_trait = lifestyle_bureaucrat }
		bureaucracy_exp_value > scope:liege.bureaucracy_exp_value
		add = 50
	}	
	#高级领主的廷臣不倾向低就
	modifier = {
		scope:liege = { is_oe_imperial_court_trigger = yes }
		add = -5
	}
	modifier = {
		any_close_family_member = {
			NOT = { this = scope:liege } #There is already a family bonus for liege
			is_landed = yes
			highest_held_title_tier < scope:liege.highest_held_title_tier
		}
		add = -50
	}

	modifier = {
		is_close_family_of = scope:liege
		add = -2
	}
	modifier = {
		any_child = {
			any_parent = {
				this = scope:liege
			}
		}
		add = -1
	}

	#高阶流官
	modifier = {
		add = 1000
		factor = governor_efficiency
		bureaucrat_level_value > scope:liege.bureaucrat_level_value
		scope:liege = { is_independent_ruler = no }
	}

	#被进献的侍臣
	modifier = {
		factor = 0
		has_character_flag = offered_courtier
	}
	
	#梁山好汉一般不离开梁山
	modifier = {
		AND = {
			has_trait = hero
			liege = { has_trait = hero }
		}
		add = -200
	}
}
