china_GetUINameNotMeNoTooltip = {
	type = character

	text = {
		localization_key = Loc_china_GetUIName_zunhao_nianhao
		trigger = {
			is_alive = yes
			has_nianhao_trigger = yes
		}
	}

	# Religion Head  转入flavoration
	text = {
		localization_key = Loc_OE_TennoGetUINameNotMeNoTooltip
		trigger = {
			culture = { has_cultural_pillar = heritage_japonic }
			dynasty ?= dynasty:japanese_yamato
			OR = {
				AND = {
					exists = primary_title
					primary_title = title:k_chrysanthemum_throne
				}
				AND = {
					exists = primary_title
					primary_title = title:e_japan
				}
				has_title = title:k_chrysanthemum_throne
				has_title = title:e_japan
			}
		}
	}
	text = {
		localization_key = Loc_china_GetUINameNotMeNoTooltip
		trigger = {
			sino_name_style_culture_trigger = yes
			OR = {
				has_miaohao_trigger = yes
				primary_title = title:h_china
			}
			NOT = { has_title = title:k_chrysanthemum_throne }
		}
	}
	
	text = {
		localization_key = Loc_china_GetUINameNoMHNotMeNoTooltip
		trigger = {
			has_miaohao_trigger = no
			has_shihao_trigger = no
			sino_name_style_culture_trigger = yes
			NOT = { has_title = title:k_chrysanthemum_throne }
		}
	}
	text = {
		localization_key = Abolished_emperor
		trigger = {
			is_alive = no
			OR = {
				has_miaohao_trigger = yes
				has_shihao_trigger = yes
			}
			OR = {
				AND = {
					exists = father
					father = {
						exists = primary_title
						primary_title = title:h_china
					}
				}
				AND = {
					exists = father.father
					father.father = {
						exists = primary_title
						primary_title = title:h_china
					}
				}
				AND = {
					any_child = {
						exists = primary_title
						primary_title = title:h_china
					}
				}
				AND = {
					exists = primary_spouse
					primary_spouse = {
						exists = primary_title
						primary_title = title:h_china
					}
				}
			}
		}
	}
	text = {
		localization_key = Loc_china_GetHistoryNameNoMHNotMeNoTooltip
		fallback = yes
	}
}
